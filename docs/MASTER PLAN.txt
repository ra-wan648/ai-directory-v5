         CHECK(free/freemium/paid), url, logo_url, logo_type(default favicon),

FILE 3: worker/worker.js  (Cloudflare Worker, D1 binding env.DB)
  Global: CORS headers (Allow-Origin:*, GET/POST/OPTIONS, JSON),
          OPTIONS preflight on all routes,
          /api/internal/* require X-Internal-Key == env.INTERNAL_API_KEY,
          cache GET via caches.default:
            /api/tools list → 10min, /api/categories → 1hr,
            /api/stats → 5min, single items → no cache
  Env vars: env.DB, env.INTERNAL_API_KEY, env.TELEGRAM_BOT_TOKEN,
            env.ADMIN_TELEGRAM_ID
  All SQL prepared statements (? placeholders). try/catch → {error} 500.
  404 for unknown routes: {error:"Not found"}.

  ROUTES (26 total):
   1  GET  /api/tools           filters: category, pricing, q, sort(newest/views/
                                  alphabetical), tag, page(default1), limit(default40)
                                  WHERE status='published'; return {tools,total,page,limit}
   2  GET  /api/tools/new       6 tools last 48hrs (tag='new' OR created_at>now-48h)
   3  GET  /api/tools/trending  ORDER views DESC, votes DESC LIMIT 6, published
   4  GET  /api/tools/featured  featured=1 AND published LIMIT 3
   5  GET  /api/tools/:slug     single + views++ + 6 related (same cat) + blog reviews
                                 → {tool, related[6], reviews[]}
   6  GET  /api/blogs           category, page, limit=12, published, ORDER published_at DESC
   7  GET  /api/blogs/:slug     single, parse faq_schema JSON before return
   8  GET  /api/prompts         category, compatible_tools, page, limit=24, published,
                                 ORDER created_at DESC
   9  GET  /api/prompts/:slug   single + copy_count++
  10  POST /api/prompts/copy/:id  copy_count+1 → {success:true}
  11  GET  /api/categories      all + recalc tool_count per category
  12  GET  /api/stats           {total_tools,total_blogs,total_prompts,
                                 total_categories,today_added}
  13  POST /api/subscribe       validate email, INSERT OR IGNORE → {success:true}
  14  POST /api/submit-tool     INSERT submitted_tools + notify admin on Telegram
                                 with [✅ Approve][❌ Reject] buttons
  15  GET  /sitemap.xml         home + /tool/{slug} + /post/{slug} + /prompts + /blog
                                 (xml, cache 1hr)
  16  GET  /rss.xml             latest 20 tools + 10 blogs (rss+xml, cache 1hr)
  17  GET  /robots.txt          User-agent:* Allow:/ Disallow:/api/internal/
                                 Sitemap: https://YOUR_DOMAIN/sitemap.xml
  18  GET  /og/tool/:slug       dynamic OG SVG image (#0d0d0d bg, name, category
                                 badge, site name; image/svg+xml, cache 24hr)
  19  GET  /tag/:tag            tools WHERE tags LIKE %tag% OR category=tag, limit40
  20  GET  /alternatives/:slug  same category excl self, ORDER views DESC LIMIT 12
                                 → {tool, alternatives[12]}
  21  GET  /compare/:slug1/:slug2  → {tool1, tool2}
  22  POST /api/internal/add-tool    dup check slug|url → update if changed (status
                                 'updated') else insert (status 'inserted');
                                 tag logic: <24hr→new, votes>50→trending, else regular
  23  POST /api/internal/add-blog    dup check slug → {duplicate} else insert pending
  24  POST /api/internal/add-prompt  insert pending → {inserted,id}
  25  POST /api/internal/bulk-insert {tools:[...]} → {inserted,updated,skipped}
  26  POST /telegram-webhook    only ADMIN_TELEGRAM_ID; callbacks:
                                 approve/reject_blog_{id}, approve/reject_prompt_{id},
                                 approve_tool_{id} (insert from submitted_tools +
                                 delete), reject_tool_{id}; commands /stats /pending;
                                 answerCallbackQuery + editMessageText after callback;
                                 wrong admin → return 200 silently.

═══════════════════════════════════════════════════════════
scripts/scrapers.py
  requests + BeautifulSoup; HEADERS UA; DELAY=2.5; AI_KEYWORDS list
  safe_scrape(url,timeout=15) → soup or None
  extract_card_data(card,source,base_url) → {name,url,raw_desc,source}
  Scrapers: toolify, futurepedia, taaft, futuretools, topaitools, aixploria,
            allthingsai, insidr, toolfk (15/12 cards each) + trendshift (2 URLs,
            AI keyword filter, stars→votes)
  scrape_all() → loops all, prints progress, total

scripts/rss.py
  feedparser; RSS_FEEDS list (producthunt AI, tldr.tech, bensbites, venturebeat,
  openai, anthropic, aiwire, marktechpost, therundown) types tools/news
  fetch_rss → 5 entries {title,url,summary[:500],type,source}; fetch_all_rss

scripts/llm.py
  uses env MANIFEST_BASE_URL + MANIFEST_API_KEY; POST {MANIFEST}/v1/chat/completions
  model "auto"; tools web_search_20250305 option; timeout 60
  extract_json(text,array) regex; parse_tool_from_raw(raw) → structured JSON
  (name,slug,desc 150-200w,short_desc,cat,pricing,url,tags) with web search,
  skip if unverifiable; generate_blog_from_news → 600-900w SEO/AEO blog with 5-FAQ,
  content HTML, focus_keyword, category; llm_find_new_tools() → array of 5 new tools;
  generate_prompt_ideas() → 3 prompts {title,slug,prompt_text,description,category,compatible_tools}

scripts/db.py
  env: CF_ACCOUNT_ID, CF_API_TOKEN, CF_D1_ID, WORKER_URL, INTERNAL_API_KEY
  D1_URL direct Cloudflare API query; d1_query(); tool_exists(slug,url);
  save_tool→POST add-tool; save_blog→POST add-blog; save_prompt→POST add-prompt;
  bulk_insert→POST bulk-insert

scripts/telegram.py
  env TELEGRAM_BOT_TOKEN, ADMIN_TELEGRAM_ID; send(method,data)
  send_blog_notification (approve_blog_{id}/reject_blog_{id} inline buttons),
  send_prompt_notification (approve/reject_prompt_{id}),
  send_error, send_summary(tools,blogs,prompts,errors)

scripts/pipeline.py
  process_tools() → scrape_all + llm_find_new_tools merged, parse via LLM, save,
                    count inserted/updated/skipped
  process_blogs() → fetch_all_rss, max 3 blogs, generate+save+notify
  process_prompts() → generate_prompt_ideas, save+notify
  main: run each in try/except (send_error on crash), send_summary at end

scripts/bulk_import.py
  bulk_progress.json checkpoint; BULK_SOURCES: toolify(100p), futurepedia(50p),
  topaitools(50p), allthingsai(30p), trendshift yearly/monthly(1p); scrape_page(30 cards);
  dup check via tool_exists, parse_tool_from_raw, batch insert every 10 via bulk_insert;
  prints Inserted/Skipped/Errors
