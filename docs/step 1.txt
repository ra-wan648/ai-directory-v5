CREATE TABLE tools (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  description TEXT,
  short_desc TEXT,
  category TEXT,
  pricing TEXT CHECK(pricing IN ('free','freemium','paid')),
  url TEXT,
  logo_url TEXT,
  logo_type TEXT DEFAULT 'favicon',
  tags TEXT,
  compatible_tools TEXT,
  views INTEGER DEFAULT 0,
  votes INTEGER DEFAULT 0,
  featured INTEGER DEFAULT 0,
  tag TEXT DEFAULT 'regular',
  status TEXT DEFAULT 'published',
  last_updated DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE blogs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  content TEXT,
  meta_description TEXT,
  focus_keyword TEXT,
  faq_schema TEXT,
  category TEXT DEFAULT 'review',
  tool_slug TEXT,
  status TEXT DEFAULT 'pending',
  telegram_message_id INTEGER,
  published_at DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE prompts (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  prompt_text TEXT NOT NULL,
  description TEXT,
  category TEXT,
  compatible_tools TEXT,
  preview_image_url TEXT,
  copy_count INTEGER DEFAULT 0,
  status TEXT DEFAULT 'pending',
  telegram_message_id INTEGER,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE categories (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT UNIQUE,
  slug TEXT UNIQUE,
  icon TEXT,
  tool_count INTEGER DEFAULT 0
);

CREATE TABLE subscribers (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  email TEXT UNIQUE NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE submitted_tools (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT,
  url TEXT,
  category TEXT,
  short_desc TEXT,
  pricing TEXT,
  submitter_email TEXT,
  status TEXT DEFAULT 'pending',
  telegram_message_id INTEGER,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

FILE 3 — worker/worker.js
═══════════════════════════════════════

Write a complete Cloudflare Worker with ALL these routes.
Use D1 binding: env.DB

CORS: All routes return these headers:
  Access-Control-Allow-Origin: *
  Access-Control-Allow-Methods: GET, POST, OPTIONS
  Content-Type: application/json

Handle OPTIONS preflight on all routes.

INTERNAL KEY: All /api/internal/* routes require header:
  X-Internal-Key: must match env.INTERNAL_API_KEY

CACHING: Cache GET responses using caches.default:
  /api/tools list → 10 min cache
  /api/categories → 1 hour cache
  /api/stats → 5 min cache
  Single item routes → no cache (views change)

─────────────────────────────
ROUTE 1: GET /api/tools
─────────────────────────────
Query params:
  category (optional, case-insensitive match)
  pricing (optional: free/freemium/paid)
  q (optional: search in name + description)
  sort (optional: newest/views/alphabetical, default: newest)
  tag (optional: new/trending/featured/regular)
  page (default: 1)
  limit (default: 40)

SQL logic:
  WHERE status='published'
  + optional filters chained with AND
  ORDER BY:
    newest → created_at DESC
    views  → views DESC
    alphabetical → name ASC
  LIMIT {limit} OFFSET {(page-1)*limit}

Also return:
  total count (for pagination)
  page, limit in response

Response:
{
  "tools": [...],
  "total": 240,
  "page": 1,
  "limit": 40
}

─────────────────────────────
ROUTE 2: GET /api/tools/new
─────────────────────────────
Return 6 tools created in last 48 hours
WHERE tag='new' OR created_at > (now - 48hrs)
ORDER BY created_at DESC
LIMIT 6

─────────────────────────────
ROUTE 3: GET /api/tools/trending
─────────────────────────────
ORDER BY views DESC, votes DESC
LIMIT 6
WHERE status='published'

─────────────────────────────
ROUTE 4: GET /api/tools/featured
─────────────────────────────
WHERE featured=1 AND status='published'
LIMIT 3

─────────────────────────────
ROUTE 5: GET /api/tools/:slug
─────────────────────────────
Get single tool by slug
Also:
  UPDATE tools SET views = views + 1 WHERE slug=?
  Get 6 related tools (same category, different slug)
  Get blog posts where tool_slug = this slug

Response:
{
  "tool": {...},
  "related": [...6 tools],
  "reviews": [...blog posts]
}

─────────────────────────────
ROUTE 6: GET /api/blogs
─────────────────────────────
Params: category, page, limit=12
WHERE status='published'
ORDER BY published_at DESC

─────────────────────────────
ROUTE 7: GET /api/blogs/:slug
─────────────────────────────
Single blog post by slug
Parse faq_schema JSON field before returning

─────────────────────────────
ROUTE 8: GET /api/prompts
─────────────────────────────
Params: category, compatible_tools, page, limit=24
WHERE status='published'
ORDER BY created_at DESC

─────────────────────────────
ROUTE 9: GET /api/prompts/:slug
─────────────────────────────
Single prompt
UPDATE copy_count++ when fetched

─────────────────────────────
ROUTE 10: POST /api/prompts/copy/:id
─────────────────────────────
UPDATE prompts SET copy_count = copy_count + 1 WHERE id=?
Return { success: true }

─────────────────────────────
ROUTE 11: GET /api/categories
─────────────────────────────
Return all categories with tool_count
Also recalculate counts:
  UPDATE categories SET tool_count = (
    SELECT COUNT(*) FROM tools
    WHERE category = categories.name
    AND status='published'
  )

─────────────────────────────
ROUTE 12: GET /api/stats
─────────────────────────────
Return:
{
  "total_tools": COUNT from tools WHERE status='published',
  "total_blogs": COUNT from blogs WHERE status='published',
  "total_prompts": COUNT from prompts WHERE status='published',
  "total_categories": COUNT from categories,
  "today_added": COUNT from tools WHERE created_at > today
}

─────────────────────────────
ROUTE 13: POST /api/subscribe
─────────────────────────────
Body: { "email": "..." }
Validate email format
INSERT OR IGNORE INTO subscribers
Return { success: true }

─────────────────────────────
ROUTE 14: POST /api/submit-tool
─────────────────────────────
Body: { name, url, category, short_desc, pricing, email }
INSERT INTO submitted_tools
Then call Telegram API to notify admin:
  Message: "🔧 New Tool Submitted!\n
  Name: {name}\nURL: {url}\nCategory: {category}"
  Buttons: [✅ Approve] [❌ Reject]
Use env.TELEGRAM_BOT_TOKEN and env.ADMIN_TELEGRAM_ID

─────────────────────────────
ROUTE 15: GET /sitemap.xml
─────────────────────────────
Generate XML sitemap with:
  / (homepage)
  /tool/{slug} for all published tools
  /post/{slug} for all published blogs
  /prompts
  /blog
Content-Type: application/xml
Cache: 1 hour

─────────────────────────────
ROUTE 16: GET /rss.xml
─────────────────────────────
Generate RSS feed with latest 20 tools + 10 blogs
Content-Type: application/rss+xml
Cache: 1 hour

─────────────────────────────
ROUTE 17: GET /robots.txt
─────────────────────────────
Return:
  User-agent: *
  Allow: /
  Disallow: /api/internal/
  Sitemap: https://YOUR_DOMAIN/sitemap.xml
Content-Type: text/plain

─────────────────────────────
ROUTE 18: GET /og/tool/:slug
─────────────────────────────
Generate dynamic OG image as SVG:
  Background: #0d0d0d
  Tool name (large, white)
  Category badge (colored)
  Site name at bottom
Content-Type: image/svg+xml
Cache: 24 hours

─────────────────────────────
ROUTE 19: GET /tag/:tag
─────────────────────────────
Return tools WHERE tags LIKE '%{tag}%'
OR category = tag
Paginated, limit 40

─────────────────────────────
ROUTE 20: GET /alternatives/:slug
─────────────────────────────
Get tool by slug
Find tools in same category (exclude this tool)
ORDER BY views DESC
LIMIT 12
Return: { tool: {...}, alternatives: [...] }

─────────────────────────────
ROUTE 21: GET /compare/:slug1/:slug2
─────────────────────────────
Get both tools by slug
Return: { tool1: {...}, tool2: {...} }

─────────────────────────────
ROUTE 22: POST /api/internal/add-tool
─────────────────────────────
Requires X-Internal-Key header
Body: tool object
Check duplicate: SELECT id FROM tools WHERE slug=? OR url=?
If exists:
  Compare pricing and description
  If changed: UPDATE tools SET ...last_updated=NOW()
  Return { status: 'updated', id }
If new:
  INSERT INTO tools
  Set tag based on:
    created_at within 24hrs → tag='new'
    votes > 50 → tag='trending'
    else → tag='regular'
  Return { status: 'inserted', id }

─────────────────────────────
ROUTE 23: POST /api/internal/add-blog
─────────────────────────────
Requires X-Internal-Key header
Body: blog object
Check duplicate: SELECT id FROM blogs WHERE slug=?
If exists: Return { status: 'duplicate' }
If new:
  INSERT INTO blogs (status='pending')
  Return { status: 'inserted', id }

─────────────────────────────
ROUTE 24: POST /api/internal/add-prompt
─────────────────────────────
Requires X-Internal-Key header
Body: prompt object
INSERT INTO prompts (status='pending')
Return { status: 'inserted', id }

─────────────────────────────
ROUTE 25: POST /api/internal/bulk-insert
─────────────────────────────
Requires X-Internal-Key header
Body: { tools: [...array of tool objects] }
Loop through each:
  Same duplicate check as add-tool
  Insert or update
Return {
  inserted: N,
  updated: N,
  skipped: N
}

─────────────────────────────
ROUTE 26: POST /telegram-webhook
─────────────────────────────
Verify update is from ADMIN_TELEGRAM_ID only.
Handle these callback_data values:

Blog callbacks:
  approve_blog_{id} → UPDATE blogs SET status='published', published_at=NOW()
  reject_blog_{id}  → DELETE FROM blogs WHERE id=?

Prompt callbacks:
  approve_prompt_{id} → UPDATE prompts SET status='published'
  reject_prompt_{id}  → DELETE FROM prompts WHERE id=?

Tool submit callbacks:
  approve_tool_{id} → INSERT into tools from submitted_tools, DELETE from submitted_tools
  reject_tool_{id}  → DELETE FROM submitted_tools WHERE id=?

Commands:
  /stats   → Send message: total tools, blogs, prompts, today added
  /pending → Send: count of pending blogs + prompts + submitted tools

After every callback:
  answerCallbackQuery (acknowledge)
  editMessageText (update original message to show result)

Error handling: if ADMIN_TELEGRAM_ID does not match, return 200 silently.

═══════════════════════════════════════
IMPORTANT NOTES FOR WORKER
═══════════════════════════════════════

1. Use env variables:
   env.DB                 ← D1 binding
   env.INTERNAL_API_KEY   ← for internal routes
   env.TELEGRAM_BOT_TOKEN ← for bot
   env.ADMIN_TELEGRAM_ID  ← for security check

2. All SQL: use prepared statements with ? placeholders

3. Error handling: wrap all DB calls in try/catch
   On error return: { error: "message" } with status 500

4. Router pattern: match pathname with if/else or URL pattern matching
   No external router library needed

5. Return 404 for unknown routes:
   { error: "Not found" } status 404
