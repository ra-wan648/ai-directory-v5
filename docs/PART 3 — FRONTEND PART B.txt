JS: js/blog.js
═══════════════════════════

For blog.html:
  activeTab = 'all'
  setupTabs(): click handlers → update activeTab → fetchBlogs()
  fetchBlogs(page=1):
    GET /api/blogs?category={activeTab}&page={page}&limit=12
    renderBlogCards(blogs)
    renderPagination(total, page)
  renderBlogCard(blog):
    Calculate read time: Math.ceil(wordCount/200)
    Format date: new Date(blog.published_at).toLocaleDateString()
    Return card HTML

For post.html:
  On load: slug = new URLSearchParams(location.search).get('slug')
  fetchPost(slug): GET /api/blogs/{slug}
  renderPost(blog):
    Set document.title = blog.title
    Set meta description
    Inject JSON-LD FAQ schema into <head>
    render content via innerHTML
    if faq_schema: renderFAQ(JSON.parse(blog.faq_schema))
    if tool_slug: fetchRelatedTool(blog.tool_slug)
    estimate and show read time
  renderFAQ(items):
    Create accordion HTML
    Toggle expand/collapse on click
    CSS max-height transition for smooth animation
  fetchRelatedTool(slug):
    GET /api/tools/{slug} → render compact tool card in sidebar
